/**
 * @file App.js
 * @description Applicazione React Native "Gioco della Sfortuna" a tema Formula 1.
 * Il giocatore deve indovinare la posizione corretta di carte in base al loro
 * livello di sfiga (da 0 a 100). Vince con 6 carte, perde con 3 errori.
 * @author Alessio Russo
 */

import React, { useState } from 'react';
import {
  SafeAreaView,
  View,
  Text,
  TouchableOpacity,
  ScrollView,
} from 'react-native';


/**
 * @typedef {Object} Carta
 * @description Rappresenta una singola carta del gioco con i suoi attributi.
 * @property {string} id       - Identificatore univoco della carta (stringa numerica, es. '1').
 * @property {string} nome     - Nome descrittivo dell'evento sfortunato.
 * @property {string} emoji    - Emoji che rappresenta visivamente la carta.
 * @property {number} sfiga    - Livello di sfortuna della carta (da 2 a 100, valori pari).
 * @property {string} desc     - Descrizione estesa dell'evento.
 */

/**
 * @typedef {'home' | 'gioco' | 'fine' | 'vittoria'} Schermata
 * @description Enumerazione delle schermate possibili dell'applicazione.
 */


/**
 * @class App
 * @description Componente root dell'applicazione "Gioco della Sfortuna".
 * Gestisce l'intero stato del gioco tramite React Hooks e il rendering
 * condizionale delle quattro schermate: home, gioco, fine e vittoria.
 * @constructs App
 * @returns {React.ReactElement} L'interfaccia dell'applicazione.
 */
export default function App() {

  /**
   * @type {[Schermata, function(Schermata): void]}
   * @description Schermata attualmente visualizzata.
   */
  const [schermata, setSchermata] = useState('home');

  /**
   * @type {[Carta[], function(Carta[]): void]}
   * @description Array di carte attualmente nella mano del giocatore,
   * sempre mantenuto ordinato per valore di sfiga crescente.
   */
  const [mano, setMano] = useState([]);

  /**
   * @type {[number, function(number): void]}
   * @description Numero di errori commessi dal giocatore nel turno corrente (0–3).
   */
  const [errori, setErrori] = useState(0);

  /**
   * @type {[Carta|null, function(Carta|null): void]}
   * @description Carta corrente da posizionare da parte del giocatore.
   * Null prima dell'inizio della partita.
   */
  const [cartaCorrente, setCartaCorrente] = useState(null);

  /**
   * @type {[string[], function(string[]): void]}
   * @description Array degli id di tutte le carte già estratte o usate,
   * per evitare duplicati durante la partita.
   */
  const [idUsati, setIdUsati] = useState([]);

  /**
   * @type {[number|null, function(number|null): void]}
   * @description Indice (0-based) della posizione scelta dal giocatore
   * nella mano per inserire la carta corrente. Null se nessuna posizione è stata scelta.
   */
  const [posScelta, setPosScelta] = useState(null);

  /**
   * @type {[boolean, function(boolean): void]}
   * @description Flag che controlla se mostrare il valore di sfiga
   * della carta corrente dopo la conferma della scelta.
   */
  const [mostraSfiga, setMostraSfiga] = useState(false);

  /**
   * @type {[string, function(string): void]}
   * @description Messaggio di feedback mostrato al giocatore dopo ogni mossa
   * (es. "Posizione corretta", "Sbagliato!", "Seleziona una posizione").
   */
  const [messaggio, setMessaggio] = useState('');



  /**
   * @function inizia
   * @description Avvia una nuova partita: estrae 3 carte iniziali per la mano
   * (ordinate per sfiga crescente) e 1 carta corrente da indovinare.
   * Reimposta tutti gli stati a zero / valori iniziali e porta il giocatore
   * alla schermata di gioco.
   * @returns {void}
   */
  function inizia() {
    const start = prendiCarte([], 3);

    const ids = start.map((c) => c.id);

    const [nuova] = prendiCarte(ids, 1);

    setMano(start.sort((a, b) => a.sfiga - b.sfiga));

    setIdUsati([...ids, nuova.id]);

    setCartaCorrente(nuova);

    setErrori(0);

    setPosScelta(null);

    setSchermata('gioco');

    setMostraSfiga(false);

    setMessaggio('');
  }

  /**
   * @function conferma
   * @description Gestisce la conferma della posizione scelta dal giocatore.
   * Valida che una posizione sia stata selezionata, inserisce la carta nella
   * posizione indicata e verifica se l'ordinamento per sfiga è rispettato.
   * - Se corretto: aggiunge la carta alla mano, mostra la sfiga e verifica
   *   la condizione di vittoria (6 carte).
   * - Se errato: incrementa il contatore errori, mostra la sfiga e verifica
   *   la condizione di sconfitta (3 errori).
   * In entrambi i casi, dopo la valutazione estrae la carta successiva.
   * @returns {void}
   */
  function conferma() {
    if (posScelta === null) {
      setMessaggio(' Seleziona una posizione');
      return;
    }

    const nuova = [...mano];

    nuova.splice(posScelta, 0, cartaCorrente);

    let ok = true;

    for (let i = 0; i < nuova.length - 1; i++) {
      if (nuova[i].sfiga > nuova[i + 1].sfiga) {
        ok = false;
      }
    }

    if (ok) {
      const nuovaMano = [...mano, cartaCorrente].sort(
        (a, b) => a.sfiga - b.sfiga
      );

      setMessaggio(' Posizione corretta');

      setMostraSfiga(true);

      setMano(nuovaMano);

      if (nuovaMano.length >= 6) {
        setSchermata('vittoria');
        return;
      }
    } else {
      const nuoviErrori = errori + 1;

      setErrori(nuoviErrori);

      setMessaggio(` Sbagliato! Sfiga: ${cartaCorrente.sfiga}`);

      setMostraSfiga(true);

      if (nuoviErrori >= 3) {
        setSchermata('fine');
        return;
      }
    }

  
    const [prossima] = prendiCarte([...idUsati, cartaCorrente.id], 1);

    if (prossima) {
      setCartaCorrente(prossima);

      setIdUsati((prev) => [...prev, cartaCorrente.id, prossima.id]);

      setPosScelta(null);

      setMostraSfiga(false);
    } else {
      setSchermata('vittoria');
    }
  }


  /**
   * @description Schermata Home: mostra titolo, istruzioni e pulsante di avvio.
   */
  if (schermata === 'home') {
    return (
      <SafeAreaView
        style={{
          flex: 1,
          backgroundColor: 'aliceblue',
        }}>
        <View
          style={{
            flex: 1,
            justifyContent: 'center',
            alignItems: 'center',
            padding: 30,
          }}>
          <Text
            style={{
              fontSize: 50,
              marginBottom: 10,
            }}></Text>

          <Text
            style={{
              fontSize: 34,
              textAlign: 'center',
              fontWeight: 'bold',
              color: 'midnightblue',
            }}>
            Gioco della
          </Text>

          <Text
            style={{
              fontSize: 40,
              textAlign: 'center',
              fontWeight: 'bold',
              color: 'deepskyblue',
              marginBottom: 20,
            }}>
            Sfortuna
          </Text>

          <Text
            style={{
              textAlign: 'center',
              fontSize: 16,
              color: 'slategray',
              marginBottom: 50,
              lineHeight: 24,
            }}>
            Indovina la posizione corretta delle carte in base al livello di
            sfiga.
          </Text>

          <TouchableOpacity
            onPress={inizia}
            style={{
              backgroundColor: 'deepskyblue',
              padding: 20,
              borderRadius: 20,
              width: '85%',
            }}>
            <Text
              style={{
                color: 'white',
                textAlign: 'center',
                fontSize: 20,
                fontWeight: 'bold',
              }}>
              Inizia Partita
            </Text>
          </TouchableOpacity>

          <View
            style={{
              marginTop: 40,
              backgroundColor: 'white',
              padding: 18,
              borderRadius: 18,
              width: '100%',
            }}>
            <Text
              style={{
                textAlign: 'center',
                color: 'gray',
                lineHeight: 22,
              }}>
              Vinci con 6 carte{'\n'}
              Perdi con 3 errori
            </Text>
          </View>
        </View>
      </SafeAreaView>
    );
  }

  /**
   * @description Schermata Gioco: mostra la mano del giocatore, la carta corrente
   * da posizionare, i pulsanti di inserimento e il pulsante di conferma.
   */
  if (schermata === 'gioco') {
    return (
      <SafeAreaView
        style={{
          flex: 1,
          backgroundColor: 'aliceblue',
        }}>
        <ScrollView>
          <View style={{ padding: 20 }}>
            <Text
              style={{
                fontSize: 22,
                fontWeight: 'bold',
                color: 'midnightblue',
              }}>
              Carte: {mano.length}/6
            </Text>

            <Text
              style={{
                color: 'crimson',
                marginTop: 6,
                fontWeight: 'bold',
              }}>
              Errori: {errori}/3
            </Text>

            {messaggio !== '' && (
              <View
                style={{
                  backgroundColor: 'white',
                  padding: 14,
                  borderRadius: 14,
                  marginTop: 15,
                }}>
                <Text
                  style={{
                    color: 'black',
                    textAlign: 'center',
                    fontWeight: 'bold',
                  }}>
                  {messaggio}
                </Text>
              </View>
            )}
          </View>

          {cartaCorrente && (
            <View
              style={{
                padding: 20,
                marginHorizontal: 20,
                backgroundColor: 'white',
                borderRadius: 20,
              }}>
              <Text
                style={{
                  fontSize: 60,
                  textAlign: 'center',
                }}>
                {cartaCorrente.emoji}
              </Text>

              <Text
                style={{
                  fontSize: 24,
                  fontWeight: 'bold',
                  textAlign: 'center',
                  color: 'midnightblue',
                  marginTop: 10,
                }}>
                {cartaCorrente.nome}
              </Text>

              <Text
                style={{
                  textAlign: 'center',
                  marginTop: 10,
                  color: 'darkorange',
                  fontSize: 18,
                  fontWeight: 'bold',
                }}>
                Sfiga: {mostraSfiga ? cartaCorrente.sfiga : '???'}
              </Text>

              <Text
                style={{
                  marginTop: 10,
                  fontSize: 16,
                  color: 'gray',
                  textAlign: 'center',
                }}>
                {cartaCorrente.desc}
              </Text>
            </View>
          )}

          <View style={{ paddingHorizontal: 20, marginTop: 20 }}>
            {mano.map((carta, index) => (
              <View key={carta.id}>
                <TouchableOpacity
                  onPress={() => setPosScelta(index)}
                  style={{
                    backgroundColor:
                      posScelta === index ? 'skyblue' : 'lightblue',
                    padding: 12,
                    borderRadius: 12,
                    marginBottom: 8,
                    alignItems: 'center',
                  }}>
                  <Text
                    style={{
                      fontWeight: 'bold',
                    }}>
                    Inserisci qui
                  </Text>
                </TouchableOpacity>

                <View
                  style={{
                    padding: 16,
                    marginBottom: 12,
                    backgroundColor: 'white',
                    borderRadius: 18,
                  }}>
                  <Text
                    style={{
                      color: 'midnightblue',
                      fontSize: 18,
                      fontWeight: 'bold',
                    }}>
                    {carta.emoji} {carta.nome}
                  </Text>

                  <Text
                    style={{
                      color: 'gray',
                      marginTop: 6,
                      fontSize: 14,
                    }}>
                    {carta.desc}
                  </Text>

                  <Text
                    style={{
                      color: 'darkorange',
                      marginTop: 8,
                      fontWeight: 'bold',
                    }}>
                    Sfiga: {carta.sfiga}
                  </Text>
                </View>
              </View>
            ))}

            <TouchableOpacity
              onPress={() => setPosScelta(mano.length)}
              style={{
                backgroundColor:
                  posScelta === mano.length ? 'skyblue' : 'lightblue',
                padding: 14,
                borderRadius: 12,
                alignItems: 'center',
                marginTop: 4,
              }}>
              <Text
                style={{
                  fontWeight: 'bold',
                }}>
                Inserisci alla fine
              </Text>
            </TouchableOpacity>
          </View>

          <Text
            style={{
              textAlign: 'center',
              marginTop: 20,
              fontSize: 18,
              color: 'midnightblue',
              fontWeight: 'bold',
            }}>
            Posizione scelta: {posScelta !== null ? posScelta + 1 : '-'}
          </Text>

          <TouchableOpacity
            onPress={conferma}
            style={{
              backgroundColor: 'deepskyblue',
              borderRadius: 18,
              padding: 20,
              margin: 20,
            }}>
            <Text
              style={{
                color: 'white',
                textAlign: 'center',
                fontWeight: 'bold',
                fontSize: 18,
              }}>
              Conferma
            </Text>
          </TouchableOpacity>
        </ScrollView>
      </SafeAreaView>
    );
  }

  /**
   * @description Schermata Fine (sconfitta): mostrata quando il giocatore
   * raggiunge 3 errori. Contiene un pulsante per tornare alla Home.
   */
  if (schermata === 'fine') {
    return (
      <SafeAreaView
        style={{
          flex: 1,
          justifyContent: 'center',
          alignItems: 'center',
          backgroundColor: 'aliceblue',
        }}>
        <Text
          style={{
            fontSize: 34,
            color: 'crimson',
            fontWeight: 'bold',
          }}>
          Hai perso
        </Text>

        <TouchableOpacity
          onPress={() => setSchermata('home')}
          style={{
            marginTop: 20,
            backgroundColor: 'deepskyblue',
            padding: 16,
            borderRadius: 12,
          }}>
          <Text
            style={{
              color: 'white',
              fontWeight: 'bold',
            }}>
            Ricomincia
          </Text>
        </TouchableOpacity>
      </SafeAreaView>
    );
  }

  /**
   * @description Schermata Vittoria: mostrata quando il giocatore colleziona
   * 6 carte corrette. Contiene un pulsante per giocare ancora.
   */
  if (schermata === 'vittoria') {
    return (
      <SafeAreaView
        style={{
          flex: 1,
          justifyContent: 'center',
          alignItems: 'center',
          backgroundColor: 'aliceblue',
        }}>
        <Text
          style={{
            fontSize: 34,
            color: 'green',
            fontWeight: 'bold',
          }}>
          Hai vinto
        </Text>

        <TouchableOpacity
          onPress={() => setSchermata('home')}
          style={{
            marginTop: 20,
            backgroundColor: 'deepskyblue',
            padding: 16,
            borderRadius: 12,
          }}>
          <Text
            style={{
              color: 'white',
              fontWeight: 'bold',
            }}>
            Gioca ancora
          </Text>
        </TouchableOpacity>
      </SafeAreaView>
    );
  }

  return null;
}

/**
 * @function prendiCarte
 * @description Estrae casualmente un numero specificato di carte dal dataset
 * globale {@link CARTE}, escludendo quelle i cui id sono già presenti
 * nell'array fornito. Le carte disponibili vengono mescolate con Fisher-Yates
 * approssimato (sort + random) prima di essere selezionate.
 * @param {string[]} idUsati - Array degli id delle carte già utilizzate/estratte,
 *   da escludere dalla selezione.
 * @param {number}   quante  - Numero di carte da estrarre.
 * @returns {Carta[]} Array di {@link Carta} estratte casualmente (lunghezza ≤ quante).
 *   Può essere più corto se le carte disponibili sono meno di quante richiesto.
 */
function prendiCarte(idUsati, quante) {
  const disponibili = CARTE.filter((c) => !idUsati.includes(c.id));

  const mescolate = [...disponibili].sort(() => Math.random() - 0.5);

  return mescolate.slice(0, quante);
}

/**
 * @function posizioneCorretta
 * @description Verifica se inserire {@link cartaNuova} alla posizione indicata
 * nella mano corrente produce un array ordinato per sfiga crescente.
 * La funzione non modifica la mano originale (opera su una copia).
 * @param {Carta[]} mano       - Array delle carte attualmente in mano al giocatore,
 *   già ordinato per sfiga crescente.
 * @param {Carta}   cartaNuova - La carta da inserire nella mano.
 * @param {number}  posizione  - Indice (0-based) in cui inserire la carta.
 *   Valori validi: da 0 (inizio) a mano.length (fine).
 * @returns {boolean} `true` se dopo l'inserimento la mano rimane ordinata
 *   per sfiga crescente, `false` altrimenti.
 */
function posizioneCorretta(mano, cartaNuova, posizione) {
  const nuova = [...mano];

  nuova.splice(posizione, 0, cartaNuova);

  for (let i = 0; i < nuova.length - 1; i++) {
    if (nuova[i].sfiga > nuova[i + 1].sfiga) {
      return false;
    }
  }

  return true;
}
